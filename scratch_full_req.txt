25: def build_tfidf_matrix(docs: list):
26:     """
27:     Convert a list of preprocessed text strings into a TF-IDF matrix.
28: 
29:     TF-IDF (Term Frequency - Inverse Document Frequency) gives higher
30:     weight to words that appear often in one document but rarely across
31:     all documents — exactly what we want for finding unique evidence links.
32: 
33:     Args:
34:         docs : List of cleaned text strings (output of nlp_engine.preprocess)
35: 
36:     Returns:
37:         matrix     : Sparse TF-IDF matrix (shape: n_docs x n_features)
38:         vectorizer : Fitted TfidfVectorizer (used to transform new docs later)
39: 
40:     Example:
41:         >>> docs = ["rajan near victim house", "meera financial dispute victim"]
42:         >>> matrix, vectorizer = build_tfidf_matrix(docs)
43:     """
44:     if not docs or len(docs) == 0:
45:         raise ValueError("Cannot build TF-IDF matrix from empty document list.")
46: 
47:     vectorizer = TfidfVectorizer(
48:         ngram_range=(1, 2),   # Use single words AND two-word phrases (bigrams)
49:                               # e.g. "financial dispute" is more meaningful than
50:                               # "financial" and "dispute" separately
51:         min_df=1,             # Include a term even if it appears in only 1 doc
52:         sublinear_tf=True,    # Apply log normalization to term frequency
53:                               # Prevents very long documents from dominating
54:     )
55: 
56:     matrix = vectorizer.fit_transform(docs)
57:     return matrix, vectorizer

60: # ── Cosine similarity ──────────────────────────────────────────────────────────
61: 
62: def cosine_similarity_score(vec_a, vec_b) -> float:
63:     """
64:     Compute cosine similarity between two TF-IDF vectors.
65: 
66:     Cosine similarity measures the angle between two vectors in
67:     high-dimensional space. A score of 1.0 means identical content,
68:     0.0 means no overlap at all.
69: 
70:     Args:
71:         vec_a : Sparse vector (1 x n_features)
72:         vec_b : Sparse vector (1 x n_features)
73: 
74:     Returns:
75:         Float between 0.0 and 1.0
76: 
77:     Example:
78:         >>> score = cosine_similarity_score(matrix[0], matrix[1])
79:         >>> print(round(score, 2))
80:         0.43
81:     """
82:     score = cosine_similarity(vec_a, vec_b)[0][0]
83:     # Clamp to [0, 1] to avoid floating point edge cases
84:     return float(np.clip(score, 0.0, 1.0))
